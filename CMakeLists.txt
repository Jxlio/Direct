# Echo cancellation, noise suppression and automatic gain for the native call path use SpeexDSP (BSD),
# a third-party library that is downloaded and compiled at configure time. It is therefore opt-in:
#   cmake -S . -B build -DMURMUR_ENABLE_SPEEXDSP=ON
option(MURMUR_ENABLE_SPEEXDSP "Build the native voice clean-up (echo, noise, gain) with SpeexDSP" OFF)
if(MURMUR_ENABLE_SPEEXDSP)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/speexdsp/libspeexdsp/mdf.c")
    set(MURMUR_SPEEXDSP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/speexdsp")
  else()
    FetchContent_Declare(murmur_speexdsp GIT_REPOSITORY https://github.com/xiph/speexdsp.git
      GIT_TAG 7a158783df74efe7c2d1c6ee8363c1e695c71226)
    FetchContent_GetProperties(murmur_speexdsp)
    if(NOT murmur_speexdsp_POPULATED)
      FetchContent_Populate(murmur_speexdsp)
    endif()
    set(MURMUR_SPEEXDSP_SOURCE "${murmur_speexdsp_SOURCE_DIR}")
  endif()
  # Echo cancellation, noise suppression and automatic gain (BSD licence, see third_party/speexdsp/COPYING.txt).
  add_library(murmur_speexdsp STATIC
    "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp/mdf.c" "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp/preprocess.c"
    "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp/fftwrap.c" "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp/filterbank.c"
    "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp/kiss_fft.c" "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp/kiss_fftr.c")
  target_include_directories(murmur_speexdsp SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/speexdsp" "${MURMUR_SPEEXDSP_SOURCE}/include")
  target_include_directories(murmur_speexdsp SYSTEM PRIVATE "${MURMUR_SPEEXDSP_SOURCE}/libspeexdsp")
  target_compile_definitions(murmur_speexdsp PRIVATE HAVE_CONFIG_H)
  if(MSVC)
    target_compile_options(murmur_speexdsp PRIVATE /W0)
  endif()
  add_library(murmur_dsp src/dsp.c)
  target_include_directories(murmur_dsp PUBLIC include)
  target_link_libraries(murmur_dsp PUBLIC murmur_speexdsp)
  if(MSVC)
    target_compile_options(murmur_dsp PRIVATE /W4 /WX /sdl)
  endif()
endif()
